# loopback packets can only be filtered using 'outbound'.
localhost ipv4 all: outbound and loopback
localhost ipv4 tcp: tcp and outbound and loopback
localhost ipv4 udp: udp and outbound and loopback

# more general
all sending packets: outbound
all receiving packets: inbound

# more specific
all ipv4 against specific ip: ip.DstAddr == 198.51.100.1 or ip.SrcAddr == 198.51.100.1
tcp ipv4 against specific ip: tcp and (ip.DstAddr == 198.51.100.1 or ip.SrcAddr == 198.51.100.1)
udp ipv4 against specific ip: udp and (ip.DstAddr == 198.51.100.1 or ip.SrcAddr == 198.51.100.1)
all ipv4 against port: tcp.DstPort == 12354 or tcp.SrcPort == 12354 or udp.DstPort == 12354 or udp.SrcPort == 12354
tcp ipv4 against port: tcp and (tcp.DstPort == 12354 or tcp.SrcPort == 12354)
udp ipv4 against port: udp and (udp.DstPort == 12354 or udp.SrcPort == 12354)

# --- Game Engine / Service presets ---

# Unreal Engine default game port
unreal engine (7777): udp and (udp.DstPort == 7777 or udp.SrcPort == 7777)

# Unity Netcode for GameObjects / Relay default port
unity netcode (9000): udp and (udp.DstPort == 9000 or udp.SrcPort == 9000)

# Unity Transport default port range
unity transport (7770-7780): udp and ((udp.DstPort >= 7770 and udp.DstPort <= 7780) or (udp.SrcPort >= 7770 and udp.SrcPort <= 7780))

# Photon Engine (Realtime/PUN2) default ports
photon engine: udp and (udp.DstPort == 5055 or udp.SrcPort == 5055 or udp.DstPort == 5056 or udp.SrcPort == 5056)

# Steam game traffic (typical UDP range)
steam game: udp and (udp.DstPort >= 27000 and udp.DstPort <= 27036)

# Steam game + query ports
steam game+query: udp and ((udp.DstPort >= 27000 and udp.DstPort <= 27050) or (udp.SrcPort >= 27000 and udp.SrcPort <= 27050))

# Epic Online Services (STUN/TURN)
epic online services: udp and (udp.DstPort == 3478 or udp.DstPort == 3479 or udp.SrcPort == 3478 or udp.SrcPort == 3479)

# Godot Engine ENet default port
godot enet (6007): udp and (udp.DstPort == 6007 or udp.SrcPort == 6007)

# Mirror Networking (Unity) default port
mirror networking (7777): tcp and (tcp.DstPort == 7777 or tcp.SrcPort == 7777)

# FishNet (Unity) default port
fishnet (7770): udp and (udp.DstPort == 7770 or udp.SrcPort == 7770)

# Minecraft (Java Edition)
minecraft java: tcp and (tcp.DstPort == 25565 or tcp.SrcPort == 25565)

# Minecraft (Bedrock Edition)
minecraft bedrock: udp and (udp.DstPort == 19132 or udp.SrcPort == 19132)
